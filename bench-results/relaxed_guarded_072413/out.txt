Here's a complete Space Invaders game using Pygame:

```python
import pygame
import random

# Initialize Pygame
pygame.init()

# Constants
SCREEN_WIDTH = 800
SCREEN_HEIGHT = 600
PLAYER_WIDTH = 60
PLAYER_HEIGHT = 40
ENEMY_WIDTH = 50
ENEMY_HEIGHT = 40
BULLET_WIDTH = 5
BULLET_HEIGHT = 15
PLAYER_SPEED = 5
ENEMY_SPEED = 2
BULLET_SPEED = 7
ENEMY_BULLET_SPEED = 4

# Colors
BLACK = (0, 0, 0)
WHITE = (255, 255, 255)
GREEN = (0, 255, 0)
RED = (255, 0, 0)
YELLOW = (255, 255, 0)

class Player:
    def __init__(self):
        self.rect = pygame.Rect(SCREEN_WIDTH // 2 - PLAYER_WIDTH // 2, 
                                SCREEN_HEIGHT - 100, 
                                PLAYER_WIDTH, PLAYER_HEIGHT)
        self.speed = PLAYER_SPEED
        self.bullets = []
        self.shoot_timer = 0
        
    def move(self, dx):
        self.rect.x += dx * self
