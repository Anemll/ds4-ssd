import pygame
import random
import math

# Initialize Pygame
pygame.init()

# Constants
SCREEN_WIDTH = 800
SCREEN_HEIGHT = 600
PLAYER_SPEED = 5
ENEMY_SPEED = 2
BULLET_SPEED = 7
ENEMY_BULLET_SPEED = 3

# Colors
BLACK = (0, 0, 0)
WHITE = (255, 255, 255)
GREEN = (0, 255, 0)
RED = (255, 0, 0)
BLUE = (0, 0, 255)
YELLOW = (255, 255, 0)

class Player:
    def __init__(self, x, y):
        self.x = x
        self.y = y
        self.width = 50
        self.height = 40
        self.speed = PLAYER_SPEED
        self.bullets = []
        self.lives = 3
        self.score = 0
        
    def move_left(self):
        self.x -= self.speed
        if self.x < 0:
            self.x = 0
            
    def move_right(self):
        self.x += self.speed
        if self.x > SCREEN_WIDTH - self.width:
            self.x = SCREEN_WIDTH - self.width
            
    def shoot(self):
        bullet = Bullet(self.x + self.width//2 - 2, self.y, BULLET_SPEED, True)
        self.bullets.append(bullet)
        
    def draw(self, screen):
        # Draw player ship (triangle shape)
        points = [
            (self.x + self.width//2, self.y),
            (self.x, self.y + self.height),
            (self.x + self.width, self.y + self.height)
        ]
        pygame.draw.polygon(screen, GREEN, points)
        
        # Draw bullets
        for bullet in self.bullets:
            bullet.draw(screen)
            
    def update(self):
        # Update bullets
        for bullet in self.bullets[:]:
            bullet.update()
            if bullet.y < 0:
                self.bullets.remove(bullet)
