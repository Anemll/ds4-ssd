import pygame

# Initialize Pygame
pygame.init()

# Constants
SCREEN_WIDTH = 800
SCREEN_HEIGHT = 600
PLAYER_WIDTH = 60
PLAYER_HEIGHT = 40
ENEMY_WIDTH = 50
ENEMY_HEIGHT = 40
BULLET_WIDTH = 5
BULLET_HEIGHT = 15
PLAYER_SPEED = 5
ENEMY_SPEED = 2

BLACK = (0, 0, 0)

class Player:
    def __init__(self, x, y):
        self.x = x
        self.y = y
        self.width = PLAYER_WIDTH
        self.height = PLAYER_HEIGHT
        self.speed = PLAYER_SPEED
        self.bullets = []
        self.lives = 3
        self.score = 0
        
    def move_left(self):
        self.x -= self.speed
        if self.x < 0:
            self.x = 0
            
    def move_right(self):
        self.x += self.speed
        if self.x > SCREEN_WIDTH - self.width:
            self.x = SCREEN_WIDTH - self.width
            
    def shoot(self):
        bullet = Bullet(self.x + self.width//2 - BULLET_WIDTH//2, 
                       self.y - BULLET_HEIGHT, 
                       BULLET_SPEED, True)
        self.bullets.append(bullet)
        
    def draw(self, screen):
        pygame.draw.rect(screen, GREEN, (self.x, self.y, self.width, self.height))
        # Draw a simple ship shape
        pygame.draw.polygon(screen, WHITE, [
            (self.x + self.width//2, self.y - 10),
            (self.x + self.width//2 - 15, self.y + 5),
            (self.x + self.width//2 + 15, self.y + 5)
        ])

class Enemy:
    def __init__(self, x, y):
        self.x = x
        self.y = y
        self.width = ENEMY_WIDTH
        self.height = ENEMY_HEIGHT
        self.speed = ENEMY_SPEED
        self.direction = 1  # 1 for right, -1 for left
        
    def move(self):
        self.x += self.speed * self.direction
        
    def draw(self, screen):
        pygame.draw.rect(screen, RED, (self.x, self.y, self.width, self.height))
        # Draw alien features
        pygame.draw.circle(screen, YELLOW, (self.x + 15, self.y + 15), 5)
        pygame.draw.circle(screen, YELLOW, (self.x + 35, self.y + 15), 5)
        pygame.draw.circle(screen, BLACK, (self.x + 15, self.y + 15), 2)
        pygame.draw.circle(screen, BLACK, (self.x + 35, self.y + 15), 2)

class Bullet:
    def __init__(self, x, y, speed, is_player_bullet=True):
        self.x = x
        self.y = y
        self.width = BULLET_WIDTH
        self.height = BULLET_HEIGHT
        self.speed = speed
        self.is_player_bullet = is_player_bullet
        
    def move(self):
        if self.is_player_bullet:
            self.y -= self.speed
        else:
            self.y += self.speed
            
    def draw(self, screen):
        color = WHITE if self.is_player_bullet else RED
        pygame.draw.rect(screen, color, (self.x, self.y, self.width, self.height))
