I'll create a complete Space Invaders game using Pygame. This will include a player ship, enemies, bullets, and scoring.

```python
import pygame
import random
import math

# Initialize Pygame
pygame.init()

# Constants
SCREEN_WIDTH = 800
SCREEN_HEIGHT = 600
PLAYER_SPEED = 5
ENEMY_SPEED = 2
BULLET_SPEED = 7
ENEMY_BULLET_SPEED = 3

# Colors
BLACK = (0, 0, 0)
WHITE = (255, 255, 255)
RED = (255, 0, 0)
GREEN = (0, 255, 0)
BLUE = (0, 0, 255)
YELLOW = (255, 255, 0)

class Player:
    def __init__(self, x, y):
        self.x = x
        self.y = y
        self.width = 50
        self.height = 40
        self.speed = PLAYER_SPEED
        self.bullets = []
        self.lives = 3
        self.score = 0
        
    def move_left(self):
        self.x -= self.speed
        if self.x < 0:
            self.x = 0
            
    def move_right(self):
        self.x += self.speed
        if self.x > SCREEN_WIDTH - self.width:
            self.x = SCREEN_WIDTH - self.width
            
    def shoot(self):
        bullet = Bullet(self.x + self.width//2 - 2, self.y, BULLET_SPEED, True)
        self.bullets.append(bullet)
        
    def draw(self, screen):
        # Draw player ship (triangle shape)
        points = [
            (self.x + self.width//2, self.y),
            (self.x, self.y + self.height),
            (self.x + self.width, self.y + self.height)
        ]
        pygame.draw.polygon(screen, GREEN, points)
        
        # Draw bullets
        for bullet in self.bullets:
            bullet.draw(screen)
            
    def update(self):
        # Update bullets
        for bullet in self.bullets[:]:
            bullet.update()
            if bullet.y < 0:
                self.bullets.remove(bullet)

class Enemy:
    def __init__(self, x, y, row, col):
        self.x = x
        self.y = y
        self.width = 40
        self.height = 30
        self.speed = ENEMY_SPEED
        self.row = row
        self.col = col
        self.direction = 1  # 1 for right, -1 for left
        self.shoot_timer = random.randint(60, 180)
        self.bullets = []
        
    def draw(self, screen):
        # Draw enemy (inverted triangle)
        points = [
            (self.x + self.width//2, self.y + self.height),
            (self.x, self.y),
            (self.x + self.width, self.y)
        ]
        pygame.draw.polygon(screen, RED, points)
        
        # Draw bullets
        for bullet in self.bullets:
            bullet.draw(screen)
            
    def update(self, enemies):
        # Move enemy
        self.x += self.speed * self.direction
        
        # Update bullets
        for bullet in self.bullets[:]:
            bullet.update()
            if bullet.y > SCREEN_HEIGHT:
                self.bullets.remove(bullet)
                
        # Random shooting
        self.shoot_timer -= 1
        if self.shoot_timer <= 0:
            self.shoot()
            self.shoot_timer = random.randint(60, 180)
            
    def shoot(self):
        bullet = EnemyBullet(self.x + self.width//2 - 2, self.y + self.height)
        self.bullets.append(bullet)

class Bullet:
    def __init__(self, x, y, speed, is_player=True):
        self.x = x
        self.y = y
        self.speed = speed
        self.width = 4
        self.height = 10
        self.is_player = is_player
        
    def draw(self, screen):
        pygame.draw.rect(screen, YELLOW, (self.x, self.y, self.width, self.height))
        
    def update(self):
        if self.is_player:
            self.y -= self.speed
        else:
            self.y += self.speed

class EnemyBullet:
    def __init__(self, x, y):
        self.x = x
